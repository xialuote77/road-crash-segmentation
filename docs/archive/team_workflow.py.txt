#  Data reading and merging
person = pd.read_csv('data/raw/person.csv')
vehicle = pd.read_csv('data/raw/vehicle.csv')
merged_df = pd.merge(person, vehicle, on=['ACCIDENT_NO', 'VEHICLE_ID'], how='inner')


selected_columns = ['HELMET_BELT_WORN', 'SEATING_POSITION', 'VEHICLE_BODY_STYLE', 'INJ_LEVEL_DESC']
df = merged_df[selected_columns].copy()


#  Feature Engineering
most_common = df["HELMET_BELT_WORN"].mode()[0]
df["HELMET_BELT_WORN"] = df["HELMET_BELT_WORN"].fillna(most_common)


def belt_status(value):
   if value == 1.0:
       return 'Worn'
   elif value == 8.0:
       return 'Not Worn'
   else:
       return 'Irrelevant'


df['HELMET_BELT_WORN_STATUS'] = df['HELMET_BELT_WORN'].apply(belt_status)


def classify_seating(seat_pos):
   if seat_pos == 'D':
       return 'Driver'
   elif seat_pos in ['LF', 'CF']:
       return 'Front Passenger'
   elif seat_pos in ['LR', 'RR', 'CR', 'OR']:
       return 'Rear Passenger'
   elif seat_pos in ['PL', 'PS']:
       return 'Motorcycle Passenger'
   else:
       return 'Unknown'

df['SEAT_CATEGORY'] = df['SEATING_POSITION'].apply(classify_seating)


def vehicle_classifier(vehicle_type):
   vehicle_type = str(vehicle_type).upper()
   if any(x in vehicle_type for x in ['SED', 'SEDAN', 'COUPE', 'CONVRT', 'RDSTR', 'WAGON', 'S WAG', 'P MVR']):
       return 'Car'
   elif any(x in vehicle_type for x in ['JEEP', 'SUV', '4WD', 'SWV1', 'SWV2']):
       return 'SUV/4WD'
   elif any(x in vehicle_type for x in ['UTE', 'DC UTE', 'UTIL', 'VAN', 'P VAN', 'TRAY', 'CARVN', 'MULTI', 'MULTIX']):
       return 'Light Commercial'
   elif any(x in vehicle_type for x in ['TRAILR', 'TR TRK', 'T TRK', 'TANKER', 'TIPPER', 'TILTER', 'C CHAS', 'T SDS', 'DUMPER', 'DOLLY', 'FLOAT', 'S TRL', 'S TANK']):
       return 'Heavy Truck'
   elif any(x in vehicle_type for x in ['SOLO', 'MOPED', 'SCOOTR', 'CYCLE', 'TRICAR', 'BIKE', 'BUGGY']):
       return 'Motorcycle'
   elif 'BUS' in vehicle_type:
       return 'Bus'
   else:
       return 'Other/Unknown'


df['VEHICLE_TYPE'] = df['VEHICLE_BODY_STYLE'].apply(vehicle_classifier)


def classify_severity(value):
   if pd.isna(value):
       return 'Unknown'
   value_clean = str(value).strip().lower()
   if value_clean in ['fatality', 'fatal injury']:
       return 'Fatal Injury'
   elif value_clean in ['serious injury']:
       return 'Serious Injury'
   elif value_clean in ['other injury']:
       return 'Other Injury'
   elif value_clean in ['not injured', 'non-injury']:
       return 'Non-Injury'
   else:
       return 'Unknown'


df['SEVERITY_CATEGORY'] = df['INJ_LEVEL_DESC'].apply(classify_severity)

df = pd.read_csv('cleaned_data.csv', usecols =(['HELMET_BELT_WORN_STATUS', 'SEAT_CATEGORY', 'VEHICLE_TYPE', 'SEVERITY_CATEGORY']))
X = df[['HELMET_BELT_WORN_STATUS', 'SEAT_CATEGORY', 'VEHICLE_TYPE']]
y = df['SEVERITY_CATEGORY']


# Check severity category distribution
print(f"Severity Distribution: ")
print(df['SEVERITY_CATEGORY'].value_counts(normalize=True))


# Encode X using OneHotEncoder
encoder = OneHotEncoder(sparse_output=False, handle_unknown="ignore")
X_encoded = encoder.fit_transform(X)
# Encode y using LabelEncoder
le = LabelEncoder()
y_encoded = le.fit_transform(y)


#Splitting data into training and testing datasets
X_train, X_test, y_train, y_test = train_test_split(X_encoded, y_encoded, test_size=0.2, random_state=42, stratify=y_encoded)


# Create and fit the linear model
# lr = Logistic Regression Model (variable name)
lr = LogisticRegression(max_iter=1000, class_weight='balanced', random_state=42)


# Perform 5-fold Stratified Cross validation for lr
cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
cv_scores_lr = cross_val_score(lr, X_encoded, y_encoded, cv=cv, scoring='accuracy')
print(f"Logistic Regression Accuracy (Cross-Validation):{np.mean(cv_scores_lr):.2f} (±{np.std(cv_scores_lr):.2f})")


# Train lr model and predict
lr.fit(X_train, y_train)
y_pred_lr = lr.predict(X_test)
# Print Classification Report for lr model
print("\nClassfication report (Logistic Regression):")
print(classification_report(y_test, y_pred_lr, target_names=le.classes_))


# Plot Confusion matrix for lr model
cm_lr = confusion_matrix(y_test, y_pred_lr)
ConfusionMatrixDisplay(confusion_matrix=cm_lr, display_labels=le.classes_).plot()
plt.title('Logistic Regression Confusion Matrix')
plt.tight_layout()
plt.show()



# Create and fit Non-linear model
# rf = RandomForestclassifier model
rf = RandomForestClassifier(n_estimators=100, max_depth=15, min_samples_leaf=10, class_weight='balanced', random_state=42)
# Perform 5-fold Stratified Cross validation for rf
cv_scores_rf = cross_val_score(rf, X_encoded, y_encoded, cv=cv, scoring='accuracy')
print(f"Random Forest Accuracy (Cross-Validation): {np.mean(cv_scores_rf):.2f} (±{np.std(cv_scores_rf):.2f})")


# Train model and predict
rf.fit(X_train, y_train)
y_pred_rf = rf.predict(X_test)


# Print Classification for rf model
print("\nClassfication report (Random Forest):")
print(classification_report(y_test, y_pred_rf, target_names=le.classes_))


# rf Confusion matrix
cm_rf = confusion_matrix(y_test, y_pred_rf)
ConfusionMatrixDisplay(confusion_matrix=cm_rf, display_labels=le.classes_).plot()
plt.title('Random Forest Confusion Matrix')
plt.tight_layout()
plt.show()
